Target Decoy FDR from COMET search results

Author

Witold Wolski

Published

November 25, 2026

Here we describe how you can estimate the false discovery rate (FDR) when matching your spectra against a FASTA sequence database using spectra search engines such as Mascot or Comet. The sequence database contains forward sequences and the same number of reverse sequences.

The data

Comet’s E-value significantly outperforms SEQUEST’s probability score and Comet’s cross-correlation score.

Define helper function

summarize_level <- function(
  data,
  level = "psm",
  score = "e.value",
  protein = "protein",
  q_value = 0.01,
  decoy = "^REV_"
) {
  fdr <- TargetDecoyFDR$new(data[[score]], decoy_hit = grepl(decoy, data[[protein]]), larger_better = FALSE)
  cutoff <- fdr$score_at(q_value = q_value, method = "fdr_simple")

  filtered <- filter(data, .data[[score]] <= .env$cutoff)
  n_decoy <- sum(grepl(decoy, filtered[[protein]]))
  n_confident <- nrow(filtered) - n_decoy

  res <- data.frame(
    n = nrow(data),
    n_confident = n_confident,
    n_decoy = n_decoy,
    fdr = n_decoy / n_confident,
    assignment_rate = nrow(filtered) / nrow(data) * 100
  )
  colnames(res) <- paste(colnames(res), level, sep = "_")
  res
}

Filter on PSM level only

Filter on PSM level and then estimate the FDR on peptide and protein level. score_cutoff() returns the score at the requested q-value; the code below prints it and keeps the PSMs that pass it.

score_cutoff <- function(
  data,
  score = "e.value",
  larger_better = FALSE,
  protein = "protein",
  q_value = 0.01,
  decoy = "^REV_"
) {
  fdr <- TargetDecoyFDR$new(data[[score]], decoy_hit = grepl(decoy, data[[protein]]), larger_better = larger_better)
  fdr$score_at(q_value = q_value, method = "fdr_simple")
}

summarize_fdrs <- function(
  data,
  filtered,
  psm_fdr_cutoff,
  peptide = "plain_peptide",
  protein = "protein",
  decoy = "^REV_"
) {
  res <- list()
  res$n_psm <- nrow(data)
  res$psm_fdr_cutoff <- psm_fdr_cutoff
  res$n_decoy_psm <- sum(grepl(decoy, filtered[[protein]]))
  res$n_confident_psm <- nrow(filtered) - res$n_decoy_psm
  res$fdr_psm <- res$n_decoy_psm / res$n_confident_psm

  # peptide level FDR
  filtered_peptides <- filtered |> select(all_of(c(peptide, protein))) |> distinct()
  res$n_decoy_peptide <- sum(grepl(decoy, filtered_peptides[[protein]]))
  res$n_confident_peptide <- nrow(filtered_peptides) - res$n_decoy_peptide
  res$fdr_peptide <- res$n_decoy_peptide / res$n_confident_peptide

  # protein level FDR
  filtered_proteins <- filtered |> select(all_of(protein)) |> distinct()
  res$n_decoy_protein <- sum(grepl(decoy, filtered_proteins[[protein]]))
  res$n_confident_protein <- nrow(filtered_proteins) - res$n_decoy_protein
  res$fdr_protein <- res$n_decoy_protein / res$n_confident_protein
  data.frame(res)
}
